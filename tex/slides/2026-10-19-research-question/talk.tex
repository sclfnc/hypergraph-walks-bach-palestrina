% The theme leaves the footline out of every dark slide. Here the dark slides of the talk carry the page number
% and the progress strip too; only the title page, which is not counted in the numbering (frame number 0), stays without. The strip that is still to come is
% nord3 on a dark slide: the light grey of the other slides would be a bright line on it.
\makeatletter
\newcommand\talk@footline{%
    \vbox{%
        \hbox to\paperwidth{\hskip\beamer@leftmargin
            \hbox to\dimexpr\paperwidth-2\beamer@leftmargin-6em\relax{%
                \usebeamerfont{source}\usebeamercolor[fg]{source}%
                \ifnum\splusnord@nsrc>0
                    \ifnum\splusnord@nsrc>1 Sources\else Source\fi: \splusnord@source
                \fi\hfill}\hfill
            {\usebeamerfont{page number current}\usebeamercolor[fg]{page number current}\insertframenumber}%
            {\usebeamerfont{page number current}\usebeamercolor[fg]{page number total}\thinspace/\thinspace}%
            {\usebeamerfont{page number total}\usebeamercolor[fg]{page number total}\inserttotalframenumber}%
            \kern0.4cm}%  % the number at the far right edge, not at the margin of the text
        \nointerlineskip\vskip3pt
        \hbox to\paperwidth{%
            {\ifsplusnord@dark\color{nord3}\else\color{nord4}\fi\vrule width\paperwidth height 3pt}%
            \hskip-\paperwidth
            {\color{nord10}\vrule height 3pt
                width \dimexpr\paperwidth*\insertframenumber/%
                    \ifnum\inserttotalframenumber>0 \inserttotalframenumber\else 1\fi\relax}%
            \hss}}}
% The title page of the theme with the title, the subtitle and the short bar under them centred; who and when stay at the bottom right.
\setbeamertemplate{title page}{%
    \global\splusnord@darktrue
    \vspace{8em}
    \begingroup
    \begin{beamercolorbox}[sep=0pt,center]{title}
        \usebeamerfont{title}\inserttitle\par
        \ifx\insertsubtitle\@empty\else
            \vskip0.25em
            {\usebeamerfont{subtitle}\usebeamercolor[fg]{subtitle}\insertsubtitle\par}
        \fi
    \end{beamercolorbox}
    \vskip1em
    \hbox to\linewidth{\hfill\color{nord10}\vrule width 3em height 3pt\hfill}
    \vskip0pt plus 1filll
    \begin{beamercolorbox}[sep=0pt,right]{author}
        \usebeamerfont{author}\insertauthor
    \end{beamercolorbox}
    \vskip0.25em
    \begin{beamercolorbox}[sep=0pt,right]{institute}
        \usebeamerfont{institute}\insertinstitute
    \end{beamercolorbox}
    \vskip0.5em
    \begin{beamercolorbox}[sep=0pt,right]{date}
        \usebeamerfont{date}\insertdate
    \end{beamercolorbox}
    \vskip\dimexpr\beamer@leftmargin-\footheight\relax
    \endgroup}
% The title of every light slide centred in its band (the theme sets it on the left).
\setbeamertemplate{frametitle}{%
    \ifsplusnord@dark\else
    \nointerlineskip
    \begin{beamercolorbox}[wd=\paperwidth,ht=0.1\paperheight,dp=0.05\paperheight,center]{frametitle}

        \rightskip=\beamer@rightmargin plus 1fil
        \parfillskip=0pt
        \usebeamerfont{frametitle}\insertframetitle\par
    \end{beamercolorbox}%
    \fi}
\setbeamertemplate{footline}{%
    \ifnum\insertframenumber=0 \ifsplusnord@dark\else\talk@footline\fi\else\talk@footline\fi}
